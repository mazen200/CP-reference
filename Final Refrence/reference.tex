\subsection{Notes for Hard questions}
\begin{itemize}
    \item Try to brute force values
    \item get the naive solution $\rightarrow$ so you can optimize it 
    \item Stress Test
    \item Test El PC in Practice
    \item Think in reverse of the problem
    \item solve the complement and total - cnt is your answer
    \item if you need to count the ways and the mean is can be computed easier 
    \item ways = expected * total
    \item Solve range? try to solve(r) - solve(l - 1)
\end{itemize}

\section{Algebra Formulas}
% ============================================================================

If $a$ and $r$ are real numbers and $r \neq 0$, then

$$\sum_{j=0}^{n} ar^j = \begin{cases} \frac{ar^{n+1}-a}{r-1} & \text{if } r \neq 1 \\ (n+1)a & \text{if } r = 1. \end{cases}$$

\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{ll}
$\sum_{k=1}^{n} k$ & $\frac{n(n+1)}{2}$ \\
$\sum_{k=1}^{n} k^2$ & $\frac{n(n+1)(2n+1)}{6}$ \\
$\sum_{k=1}^{n} k^3$ & $\frac{n^2(n+1)^2}{4}$ \\
$\sum_{k=0}^{\infty} x^k, \|x\| < 1$ & $\frac{1}{1-x}$ \\
$\sum_{k=1}^{\infty} kx^{k-1}, \|x\| < 1$ & $\frac{1}{(1-x)^2}$ \\
\end{tabular}
\end{center}

\begin{itemize}
    \item \textbf{LCM * GCD} = the multiple of the 2 numbers
    \item \textbf{LCM} = multiply the numbers which has the greatest power from the 2 numbers even if they're not common
    \item \textbf{GCD} = multiply the numbers which has the lowest power from the 2 numbers even if they're not common
    \item \textbf{Arithmetic sequence :} $\frac{n}{2}(a + b)$
    \item \textbf{Geometric sequence :} $\frac{L*R-a}{r-1} = a * \frac{r^n-1}{r-1}$
    \item \textbf{Sum of odd numbers :} $\left(\frac{n+n\%2}{2}\right)^2$
    \item \textbf{Sum of even numbers:} $\frac{\left(\frac{n}{2}*(2+n-n\%2)\right)}{2}$
    \item \textbf{Quadratic Formula :} $\frac{-b\pm\sqrt{b^2-4ac}}{2a}$
\end{itemize}
